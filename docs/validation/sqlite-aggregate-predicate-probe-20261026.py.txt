from pathlib import Path
import subprocess,tempfile,shutil,json,hashlib,re,random
root=Path('/dev/shm/sqleanite-json-valid-20261026');c='/dev/shm/sqleanite-tools-20261024/bin/sqlite3';lean=str(root/'.lake/build/bin/sqleanite');rng=random.Random(354941)
def run(cmd,sql):
 r=subprocess.run(cmd,input=sql.encode(),capture_output=True,timeout=30);return dict(code=r.returncode,stdout=r.stdout.decode('latin1'),stderr=r.stderr.decode('latin1'))
def err(r):return re.sub(r'^(?:Parse error|Runtime error|Error)(?: near line [0-9]+)?: ?', '',r['stderr'].splitlines()[0]) if r['code'] and r['stderr'] else ''
def csql(path,sql):return run([c,'-batch','-bail',str(path)],sql)
setup='CREATE TABLE t(id INTEGER PRIMARY KEY,v);CREATE TABLE u(id INTEGER PRIMARY KEY,k);'+''.join(f'INSERT INTO t VALUES({i},{rng.randrange(-100,100)});INSERT INTO u VALUES({i},{i%3});' for i in range(1,9))
queries=[]
for bad in [1,3,8]:
 for pred in [1,3,8]:
  value=f"CASE WHEN id={bad} THEN 'x' ELSE v END"
  predicate=f'CASE WHEN id={pred} THEN abs(-9223372036854775808) ELSE 1 END'
  for expr in [f'median({value})',f'median({value} ORDER BY id)',f'percentile(v,NULL)',f'sum(CASE WHEN id={bad} THEN abs(-9223372036854775808) ELSE v END)',f'median({value}),sum(v)']:
   for group in ['', ' GROUP BY 1+0']:
    queries.append(f'SELECT {expr} FROM t WHERE {predicate}{group};')
cases=[]
with tempfile.TemporaryDirectory(prefix='sqleanite-predicate-probe-') as td:
 p=Path(td);base=p/'base.db';assert csql(base,setup)['code']==0
 for mode in ['--verified-file','--verified-wal-file']:
  left=p/(mode+'lean.db');right=p/(mode+'c.db');shutil.copyfile(base,left);shutil.copyfile(base,right)
  for sql in queries:
   l=run([lean,mode,str(left)],sql);r=csql(right,sql);cases.append(dict(mode=mode,sql=sql,lean=l,sqlite=r,matched=bool(l['code'])==bool(r['code']) and l['stdout']==r['stdout'] and err(l)==err(r)))
h=hashlib.sha256()
for f in sorted((root/'SQLeanite').rglob('*.lean'))+[root/'Main.lean']:h.update(str(f.relative_to(root)).encode()+b'\0'+f.read_bytes())
out=dict(seed=354941,setup=setup,cases=cases,failures=sum(not x['matched'] for x in cases),implementation_sha256=h.hexdigest(),executable_sha256=hashlib.sha256(Path(lean).read_bytes()).hexdigest(),harness_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest())
Path('/tmp/sqleanite-aggregate-predicate-probe.json').write_text(json.dumps(out,indent=2)+'\n')
print('cases',len(cases),'failures',out['failures'])
for x in [x for x in cases if not x['matched']][:4]:print(x['sql'],err(x['lean']),err(x['sqlite']))
