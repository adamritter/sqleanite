from pathlib import Path
import hashlib,json,random,re,subprocess,time
root=Path('/dev/shm/sqleanite-catalog-20261026')
prototype=Path('/tmp/CtasCatalogPrototype.lean').read_text()
keywords=json.loads(re.search(r'def keywords : List String := (\[.*\])',prototype).group(1))
rng=random.Random(355361)
inputs=[]
def add(group,name,columns): inputs.append(dict(group=group,name=name,columns=columns))
affinities=['blob','integer','numeric','real','text']
for name in keywords+['within','Within','normal','abc123','_','', '0','0xAB','é','中','a$b','a b','a"b','a\nb','a;b']:
 for affinity in affinities:
  add('keyword-and-identifier',name,[('value',affinity)])
  add('keyword-and-column','made',[(name,affinity)])
for size in range(1,66):
 for char in ['a','"','é']:
  add('layout-threshold','t',[(char*size,'blob')])
  add('layout-threshold',char*size,[('v','real')])
for _ in range(2048):
 name=''.join(rng.choice('abAZ01_ $";\né中') for _ in range(rng.randrange(1,50)))
 columns=[]
 for i in range(rng.randrange(1,12)):
  column=''.join(rng.choice('abAZ01_ $";\né中') for _ in range(rng.randrange(0,40)))+'_'+str(i)
  columns.append((column,rng.choice(affinities)))
 add('seeded-identifier-layout',name,columns)
def quote(name):return '"'+name.replace('"','""')+'"'
expressions=dict(blob='1',integer='CAST(1 AS INTEGER)',numeric='CAST(1 AS NUMERIC)',real='CAST(1 AS REAL)',text='CAST(1 AS TEXT)')
for case in inputs:
 sql='CREATE TABLE '+quote(case['name'])+' AS SELECT '+','.join(expressions[a]+' AS '+quote(n) for n,a in case['columns'])+';SELECT hex(sql) FROM sqlite_schema WHERE type=\'table\';'
 result=subprocess.run(['/dev/shm/sqleanite-tools-20261024/bin/sqlite3','-batch','-bail',':memory:',sql],capture_output=True,text=True)
 assert result.returncode==0,(sql,result.stderr)
 case.update(sql=sql,sqlite=result.stdout.strip())
items=['('+json.dumps(c['name'])+', ['+', '.join('('+json.dumps(n)+', .'+a+')' for n,a in c['columns'])+'])' for c in inputs]
blocks=[]
for i in range(0,len(items),160):
 blocks.append('def inputs'+str(i)+' : List (String × List (String × SQLeanite.SQLite.Affinity)) := [\n'+',\n'.join(items[i:i+160])+']\n')
source=prototype+'\n'+'\n'.join(blocks)+'\n#eval do\n  for batch in ['+', '.join('inputs'+str(i) for i in range(0,len(items),160))+'] do\n    for (name, columns) in batch do\n      IO.println (SQLeanite.SQLite.Text.hexBytes (SQLeanite.SQLite.CtasCatalog.print name columns).toUTF8.data)\n'
leanfile=Path('/tmp/CtasCatalogCases.lean');leanfile.write_text(source)
start=time.time();run=subprocess.run(['lake','env','lean',str(leanfile)],cwd=root,capture_output=True,text=True)
lines=run.stdout.splitlines()
if run.returncode or len(lines)!=len(inputs):
 print(run.stdout);print(run.stderr)
else:
 for case,line in zip(inputs,lines):case.update(lean=line,matched=line==case['sqlite'])
report=dict(seed=355361,cases=inputs,failures=sum(not c.get('matched',False) for c in inputs),lean_exit_code=run.returncode,stdout=run.stdout if run.returncode else None,stderr=run.stderr,seconds=time.time()-start,prototype_sha256=hashlib.sha256(prototype.encode()).hexdigest(),scope='Development printer component only. Names and affinities supplied explicitly; no SQL admission, CTAS affinity inference or file publication.')
Path('/tmp/sqleanite-catalog-ctas-component-initial.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(dict(cases=len(inputs),failures=report['failures'],lean_exit_code=run.returncode)))
raise SystemExit(bool(report['failures']))
