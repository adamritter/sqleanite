from pathlib import Path
f=Path('/tmp/JsonPrettyPrototype.lean');s=f.read_text()
s=s.replace('  data : Array UInt8\n  parseDepth : Nat','  data : List (Array UInt8)\n  parseDepth : Nat')
s=s.replace('(["null","true","false"].getD kind "").toUTF8.data, depth','[( ["null","true","false"].getD kind "").toUTF8.data], depth')
s=s.replace('⟨finish, data, depth⟩','⟨finish, [data], depth⟩').replace('⟨finish, ← hexNumber data, depth⟩','⟨finish, [← hexNumber data], depth⟩').replace('⟨finish, ← float5 data, depth⟩','⟨finish, [← float5 data], depth⟩')
s=s.replace('⟨finish, #[34] ++ data ++ #[34], depth⟩','⟨finish, [#[34], data, #[34]], depth⟩')
s=s.replace('⟨finish, #[34] ++ (← text5 data (data.size+1) 0 #[]) ++ #[34], depth⟩','⟨finish, [#[34], (← text5 data (data.size+1) 0 #[]), #[34]], depth⟩')
s=s.replace('⟨finish, quoted data, depth⟩','⟨finish, [quoted data], depth⟩').replace('⟨finish, #[opening,closing], depth⟩','⟨finish, [#[opening,closing]], depth⟩')
s=s.replace('start finish #[]','start finish []').replace('start finish 0 #[]','start finish 0 []')
s=s.replace('#[opening,10] ++ children.data ++ #[10] ++ indentation indent level ++ #[closing]','[#[opening,10]] ++ children.data ++ [#[10], indentation indent level, #[closing]]')
s=s.replace('#[opening] ++ children.data ++ #[closing]','[#[opening]] ++ children.data ++ [#[closing]]')
s=s.replace('Nat → Array UInt8 → Except String Rendered','Nat → List (Array UInt8) → Except String Rendered')
s=s.replace('(out ++ sep ++ child.data)','(out ++ [sep] ++ child.data)').replace('out ++ indentation indent level','out ++ [indentation indent level]').replace('out ++ key.data ++ #[58,32]','out ++ key.data ++ [#[58,32]]').replace('(out ++ #[44,10])','(out ++ [#[44,10]])')
s=s.replace('Text.fromBytes rendered.data','Text.fromBytes (rendered.data.foldl (fun out chunk => out ++ chunk) #[])')
Path('/tmp/JsonPrettyOptimized.lean').write_text(s)
h=Path('/tmp/sqleanite-json-pretty-component.py').read_text().replace('/tmp/JsonPrettyPrototype.lean','/tmp/JsonPrettyOptimized.lean').replace('/tmp/JsonPrettyComponent.lean','/tmp/JsonPrettyOptimizedComponent.lean').replace('/tmp/sqleanite-json-pretty-component-initial.json','/tmp/sqleanite-json-pretty-component-optimized.json').replace('/tmp/sqleanite-json-pretty-component.stdout','/tmp/sqleanite-json-pretty-component-optimized.stdout').replace('/tmp/sqleanite-json-pretty-component.stderr','/tmp/sqleanite-json-pretty-component-optimized.stderr')
Path('/tmp/sqleanite-json-pretty-optimized-component.py').write_text(h)
